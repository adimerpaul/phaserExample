Agrupar objetos 
// Group: agrupa varios Mesh como uno solo
// Mover/rotar el grupo mueve todo adentro
const arbol = new THREE.Group();
arbol.add(tronco);
arbol.add(copa);
arbol.position.set(3, 0, 2);
scene.add(arbol);
Cámara en 3ª persona 
// En animate(), hacer que la cámara siga al jugador:
const offset = new THREE.Vector3(0, 4, 8);

// Posición de la cámara = jugador + offset
camera.position.copy(
  jugador.position
).add(offset);

// Siempre mirar al jugador
camera.lookAt(jugador.position);
Niebla (atmosphere) 
// Fog(color, near, far)
// near = donde empieza la niebla
// far  = donde los objetos desaparecen
scene.fog = new THREE.Fog(
  0x87ceeb, // color del cielo
  10,
  50
);

// El background debe ser el mismo color
scene.background = new THREE.Color(0x87ceeb);
Ciclo día/noche 
// Cambiar la intensidad de la luz con el tiempo
const reloj = new THREE.Clock();

function animate() {
  const t = reloj.getElapsedTime();
  // sin() oscila entre -1 y 1
  // mapear a rango de intensidad (0.1 a 1.5)
  const brillo = (Math.sin(t * 0.3) + 1) / 2;
  luzSol.intensity = brillo * 1.5 + 0.1;
}
