FBXLoader vs GLTFLoader

import { FBXLoader }

// GLB/GLTF: formato moderno recomendado
import { GLTFLoader }

// Mixer = "reproductor" de animaciones
// Uno por personaje
mixer = new THREE.AnimationMixer(personaje);

// Registrar cada clip como "acción"
fbx.animations.forEach(clip => {
  acciones[clip.name] = mixer.clipAction(clip);
});

// En animate() actualizar el mixer:

Cambiar animación con crossFade 
function cambiarAnim(nombre) {
  const nueva = acciones[nombre];
  if (accionActual === nueva) return;

  nueva.reset();
  nueva.play();

  // crossFadeTo: transición suave
  // (0.3 = duración en segundos)
  accionActual.crossFadeTo(nueva, 0.3);
  accionActual = nueva;
}
