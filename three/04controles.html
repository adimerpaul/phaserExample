OrbitControls para rotar la cámara con el mouse. WASD para mover un objeto con el teclado.

OrbitControls (mouse) 
import { OrbitControls } from
  'three/examples/jsm/controls/OrbitControls.js';

const controls = new OrbitControls(
  camera,
  renderer.domElement
);

// En el loop animate():
controls.update(); // ← obligatorio

// Opciones útiles:
controls.enableDamping = true; // inercia
controls.autoRotate = true;    // gira solo
Teclado con keydown/keyup 
// Guardar qué teclas están presionadas
const teclas = {};

window.addEventListener('keydown', e => {
  teclas[e.key] = true;  // presionada
});
window.addEventListener('keyup', e => {
  teclas[e.key] = false; // soltada
});

// En animate():
if (teclas['w']) objeto.position.z -= 0.1;
if (teclas['s']) objeto.position.z += 0.1;
Velocidad y delta time 
// El problema: en PCs rápidas el objeto
// se mueve más rápido (más frames/seg)

// Solución: usar el tiempo entre frames
const clock = new THREE.Clock();

function animate() {
  const delta = clock.getDelta(); // seg
  // Velocidad independiente del framerate:
  objeto.position.x += 2 * delta; // 2 unidades/seg
}
Mover la cámara 
// Seguir al jugador (cámara en tercera persona)
camera.position.x = jugador.position.x;
camera.position.z = jugador.position.z + 5;
camera.position.y = jugador.position.y + 3;
camera.lookAt(jugador.position);
